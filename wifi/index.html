server.js (Node.js Backend Local IP Auto-Detector) 
const express = require('express');
const http = require('http');
const os = require('os');
const path = require('path');
const WebSocket = require('ws');

const app = express();
const server = http.createServer(app);
const wss = new WebSocket.Server({ server });

const PORT = 3000;

// Automatically detect Local IPv4 Address (e.g. 192.168.x.x)
function getLocalIpAddress() {
    const interfaces = os.networkInterfaces();
    for (const name of Object.keys(interfaces)) {
        for (const iface of interfaces[name]) {
            // Skip over internal (i.e. 127.0.0.1) and non-ipv4 addresses
            if (iface.family === 'IPv4' && !iface.internal) {
                return iface.address;
            }
        }
    }
    return '127.0.0.1';
}

// Serve static HTML/JS frontend
app.use(express.static(__dirname));

app.get('/', (req, res) => {
    res.sendFile(path.join(__dirname, 'index.html'));
});

// WebSocket relay for local peer text & file signaling
wss.on('connection', (ws) => {
    console.log('[+] New device connected on Local Wi-Fi');

    ws.on('message', (message) => {
        // Broadcast incoming message/file chunk to all connected devices on LAN
        wss.clients.forEach((client) => {
            if (client !== ws && client.readyState === WebSocket.OPEN) {
                client.send(message.toString());
            }
        });
    });

    ws.on('close', () => console.log('[-] Device disconnected'));
});

// Start Server
const localIP = getLocalIpAddress();
server.listen(PORT, '0.0.0.0', () => {
    console.log('\n==================================================');
    console.log('🚀 LOCAL WI-FI SHARE SERVER IS RUNNING!');
    console.log('==================================================');
    console.log(`\n💻 On PC Open:    http://localhost:${PORT}`);
    console.log(`📱 On Phone Open: http://${localIP}:${PORT}`);
    console.log('\n(Ensure both PC and Phone are on the SAME Wi-Fi network)\n');
});

    <script>
        // Connect to WebSocket Server on same local IP
        const protocol = location.protocol === 'https:' ? 'wss:' : 'ws:';
        const ws = new WebSocket(`${protocol}//${location.host}`);

        ws.onmessage = (event) => {
            const data = JSON.parse(event.data);
            const feed = document.getElementById('feed');

            if (data.type === 'text') {
                const item = document.createElement('div');
                item.className = 'bg-slate-950 border border-slate-800 rounded-xl p-3 text-sm flex justify-between items-center';
                item.innerHTML = `<span class="text-slate-200">${data.text}</span><button onclick="navigator.clipboard.writeText('${data.text}')" class="text-xs bg-slate-800 px-2 py-1 rounded text-slate-300">Copy</button>`;
                feed.appendChild(item);
            } else if (data.type === 'file') {
                const item = document.createElement('div');
                item.className = 'bg-slate-950 border border-slate-800 rounded-xl p-3 text-sm flex justify-between items-center';
                item.innerHTML = `<span class="text-emerald-400 font-semibold">${data.filename}</span><a href="${data.fileData}" download="${data.filename}" class="text-xs bg-emerald-600 px-3 py-1 rounded font-medium text-white">Download</a>`;
                feed.appendChild(item);
            }
        };

        function sendTextMessage() {
            const input = document.getElementById('text-msg');
            if (!input.value) return;
            ws.send(JSON.stringify({ type: 'text', text: input.value }));
            input.value = '';
        }

        function sendFile(file) {
            if (!file) return;
            const reader = new FileReader();
            reader.onload = () => {
                ws.send(JSON.stringify({
                    type: 'file',
                    filename: file.name,
                    fileData: reader.result
                }));
            };
            reader.readAsDataURL(file);
        }
    </script>
